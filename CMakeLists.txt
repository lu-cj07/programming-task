cmake_minimum_required(VERSION 3.16)
project(programming_task LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(ring_buffer_test tests/ring_buffer_test.cpp)
target_include_directories(ring_buffer_test PRIVATE include)
if(MSVC)
  target_compile_options(ring_buffer_test PRIVATE /W4)
else()
  target_compile_options(ring_buffer_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME ring_buffer_test COMMAND ring_buffer_test)
