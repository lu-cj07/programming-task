cmake_minimum_required(VERSION 3.16)
project(programming_task LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()

add_executable(ring_buffer_test tests/ring_buffer_test.cpp)
target_include_directories(ring_buffer_test PRIVATE include)
add_executable(timestamp_index_test tests/timestamp_index_test.cpp)
target_include_directories(timestamp_index_test PRIVATE include)
add_executable(sensor_log_test tests/sensor_log_test.cpp)
target_include_directories(sensor_log_test PRIVATE include)

foreach(test_target IN ITEMS ring_buffer_test timestamp_index_test sensor_log_test)
  if(MSVC)
    target_compile_options(${test_target} PRIVATE /W4)
  else()
    target_compile_options(${test_target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()

add_test(NAME ring_buffer_test COMMAND ring_buffer_test)
add_test(NAME timestamp_index_test COMMAND timestamp_index_test)
add_test(NAME sensor_log_test COMMAND sensor_log_test)
